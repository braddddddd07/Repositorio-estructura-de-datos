import tkinter as tk
from tkinter import messagebox, ttk

class PilaGUI:
    def __init__(self, root):
        self.root = root
        self.root.title("Simulador Gráfico de Pila (Stack) - Estructuras Lineales")
        self.root.geometry("850x600")
        self.root.configure(bg="#0f172a") # Slate-900

        # Propiedades de la Pila
        self.capacidad_maxima = 8
        self.pila = []

        self.crear_interfaz()

    def crear_interfaz(self):
        # Título Principal
        titulo = tk.Label(
            self.root, 
            text="Estructura de Datos: PILA (LIFO)", 
            font=("Helvetica", 16, "bold"), 
            bg="#0f172a", 
            fg="#f8fafc"
        )
        titulo.pack(pady=10)

        # Contenedor Principal (2 Columnas)
        main_frame = tk.Frame(self.root, bg="#0f172a")
        main_frame.pack(fill=tk.BOTH, expand=True, padx=15, pady=5)

        # --- COLUMNA IZQUIERDA: Controles y Registro ---
        left_frame = tk.Frame(main_frame, bg="#1e293b", padx=15, pady=15, bd=1, relief=tk.SOLID)
        left_frame.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=(0, 10))

        # Entrada de datos
        tk.Label(left_frame, text="Valor a Apilar (Push):", font=("Helvetica", 10, "bold"), bg="#1e293b", fg="#cbd5e1").pack(anchor="w")
        self.entry_val = tk.Entry(left_frame, font=("Helvetica", 11), bg="#0f172a", fg="#ffffff", insertbackground="white")
        self.entry_val.pack(fill=tk.X, pady=(2, 10))
        self.entry_val.bind("", lambda e: self.push())

        # Botones de Operación
        btn_push = tk.Button(left_frame, text="PUSH (Apilar)", bg="#4f46e5", fg="white", font=("Helvetica", 10, "bold"), command=self.push, cursor="hand2")
        btn_push.pack(fill=tk.X, pady=3)

        btn_pop = tk.Button(left_frame, text="POP (Desapilar)", bg="#e11d48", fg="white", font=("Helvetica", 10, "bold"), command=self.pop, cursor="hand2")
        btn_pop.pack(fill=tk.X, pady=3)

        btn_peek = tk.Button(left_frame, text="PEEK / TOP (Ver Tope)", bg="#d97706", fg="white", font=("Helvetica", 10, "bold"), command=self.peek, cursor="hand2")
        btn_peek.pack(fill=tk.X, pady=3)

        btn_clear = tk.Button(left_frame, text="CLEAR (Vaciar Pila)", bg="#475569", fg="white", font=("Helvetica", 10), command=self.clear, cursor="hand2")
        btn_clear.pack(fill=tk.X, pady=3)

        # Registro / Log de Movimientos
        tk.Label(left_frame, text="Historial de Movimientos:", font=("Helvetica", 10, "bold"), bg="#1e293b", fg="#cbd5e1").pack(anchor="w", pady=(15, 2))
        
        self.log_listbox = tk.Listbox(left_frame, font=("Consolas", 9), bg="#0f172a", fg="#38bdf8", selectbackground="#334155")
        self.log_listbox.pack(fill=tk.BOTH, expand=True)

        # --- COLUMNA DERECHA: Canvas Gráfico de la Pila ---
        right_frame = tk.Frame(main_frame, bg="#1e293b", padx=15, pady=15, bd=1, relief=tk.SOLID)
        right_frame.pack(side=tk.RIGHT, fill=tk.BOTH, expand=True)

        tk.Label(right_frame, text="Representación Gráfica de la Pila", font=("Helvetica", 11, "bold"), bg="#1e293b", fg="#f8fafc").pack(pady=(0, 10))

        # Canvas para dibujar los elementos de la pila
        self.canvas = tk.Canvas(right_frame, bg="#0f172a", highlightthickness=0)
        self.canvas.pack(fill=tk.BOTH, expand=True)

        self.log_movimiento("SISTEMA: Pila inicializada correctamente (Capacidad: 8)")
        self.dibujar_pila()

    def log_movimiento(self, texto):
        """Agrega un mensaje con formato al historial de movimientos"""
        self.log_listbox.insert(tk.END, f"> {texto}")
        self.log_listbox.see(tk.END)

    def push(self):
        valor = self.entry_val.get().strip()
        if not valor:
            messagebox.showwarning("Atención", "Por favor ingresa un valor válido.")
            return

        if len(self.pila) >= self.capacidad_maxima:
            self.log_movimiento(f"ERROR: Overflow / Desbordamiento al intentar añadir '{valor}'")
            messagebox.showerror("Pila Llena", "¡Error de Desbordamiento (Overflow)! La pila ha alcanzado su capacidad máxima.")
            return

        self.pila.append(valor)
        self.entry_val.delete(0, tk.END)
        self.log_movimiento(f"PUSH: Se agregó el elemento '{valor}' en el tope (Pos: {len(self.pila)-1})")
        self.dibujar_pila()

    def pop(self):
        if not self.pila:
            self.log_movimiento("ERROR: Underflow / Pila vacía al intentar hacer POP")
            messagebox.showwarning("Pila Vacía", "¡Error de Subdesbordamiento (Underflow)! No hay elementos para desapilar.")
            return

        elemento_removido = self.pila.pop()
        self.log_movimiento(f"POP: Se retiró el elemento '{elemento_removido}'")
        self.dibujar_pila()

    def peek(self):
        if not self.pila:
            self.log_movimiento("PEEK: Intento de consulta en Pila Vacía")
            messagebox.showinfo("Peek / Top", "La pila se encuentra actualmente VACÍA.")
            return

        tope = self.pila[-1]
        self.log_movimiento(f"PEEK: El elemento en el tope actual es '{tope}'")
        messagebox.showinfo("Peek / Top", f"Elemento en el tope superior: {tope}")

    def clear(self):
        if not self.pila:
            return
        self.pila.clear()
        self.log_movimiento("CLEAR: Se ha vaciado la pila por completo.")
        self.dibujar_pila()

    def dibujar_pila(self):
        """Redibuja dinámicamente los elementos dentro del canvas"""
        self.canvas.delete("all")
        width = self.canvas.winfo_width()
        height = self.canvas.winfo_height()

        if width <= 1: # Ajuste inicial si aún no se renderiza completamente
            width = 350
            height = 380

        # Dibujar estructura contenedora (U shape)
        box_width = 180
        x1 = (width - box_width) / 2
        x2 = x1 + box_width
        y_bottom = height - 30
        box_height = 300
        y_top = y_bottom - box_height

        # Contorno de la pila
        self.canvas.create_line(x1, y_top, x1, y_bottom, fill="#6366f1", width=4)
        self.canvas.create_line(x1, y_bottom, x2, y_bottom, fill="#6366f1", width=4)
        self.canvas.create_line(x2, y_bottom, x2, y_top, fill="#6366f1", width=4)

        # Dibujar elementos existentes
        block_height = box_height / self.capacidad_maxima
        padding = 4

        for index, item in enumerate(self.pila):
            item_y2 = y_bottom - (index * block_height) - padding
            item_y1 = item_y2 - block_height + (padding * 2)

            # Resaltar si es el elemento TOP
            es_top = (index == len(self.pila) - 1)
            color_fondo = "#f59e0b" if es_top else "#3b82f6"

            # Dibujar el bloque
            self.canvas.create_rectangle(
                x1 + padding + 4, item_y1, 
                x2 - padding - 4, item_y2, 
                fill=color_fondo, outline="#ffffff", width=1.5
            )
            
            # Texto dentro del bloque
            self.canvas.create_text(
                (x1 + x2) / 2, (item_y1 + item_y2) / 2, 
                text=str(item), fill="white", font=("Helvetica", 11, "bold")
            )

            # Etiqueta de indicador "TOP ->"
            if es_top:
                self.canvas.create_text(
                    x1 - 35, (item_y1 + item_y2) / 2, 
                    text="TOP ➔", fill="#f59e0b", font=("Helvetica", 10, "bold")
                )

if __name__ == "__main__":
    root = tk.Tk()
    app = PilaGUI(root)
    root.mainloop()
